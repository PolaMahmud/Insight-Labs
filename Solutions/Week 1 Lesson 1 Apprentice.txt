import pandas as pd

df = pd.read_csv("sports_day.csv")

# Inspect the type of every column
print("Data types:")
print(df.dtypes)

# Classify the qualitative vs quantitative columns
print("House: Qualitative")
print("Age: Quantitative")

# The trap: the sprint column is text because of the injected "s"
sprint = "100m Sprint Time (seconds)"
print("Sprint dtype:", df[sprint].dtype)   # -> object

# Prove you cannot average a column of strings
try:
    print(df[sprint].mean())
except TypeError as e:
    print("TypeError:", e)
