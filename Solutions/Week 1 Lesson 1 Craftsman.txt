import pandas as pd

df = pd.read_csv("sports_day.csv")
col = "100m Sprint Time (seconds)"

# Step 1: strip the lowercase "s" off the sprint times
df[col] = df[col].str.replace("s", "", regex=False)

# Step 2: cast the cleaned text to float (NOT int — keep decimals)
df[col] = df[col].astype(float)

# Step 3: verify the maths now works
print("New data types:")
print(df.dtypes)
print(f"Mean: {df[col].mean():.2f}s")
print(f"Max:  {df[col].max():.2f}s")
print(f"Min:  {df[col].min():.2f}s")
